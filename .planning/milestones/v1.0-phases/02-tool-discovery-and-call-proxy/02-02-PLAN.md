    **Change 4: Add the handler methods with sessionId guard**

    IMPORTANT (addresses review concern - HIGH): Both `handleUpstreamToolsList` and `handleUpstreamToolCall`
    MUST guard against undefined sessionId BEFORE using the non-null assertion. Without this guard,
    stdio transport (where sessionId is undefined) would throw a runtime TypeError.

    ```typescript
    private async handleUpstreamToolsList(
      req: Record<string, unknown>,
      sessionId: string | undefined,
      profileId: string | undefined,
      provider: UpstreamMcpServerConfig,
    ): Promise<unknown> {
      // Guard: upstream_mcp requires session context (HTTP transport only)
      if (!sessionId) {
        throw new UpstreamConnectionError(
          'upstream_mcp requires a session context (HTTP transport only)',
          { correlationId: generateCorrelationId() }
        );
      }
      const token = this.getUpstreamToken(sessionId, profileId);
      const client = await this.getUpstreamClientFn!(sessionId, provider, token);
      const result = await client.listTools();
      const sanitized = sanitizeToolList(result.tools, this.logger);
      return {
        jsonrpc: '2.0',
        id: (req as Record<string, unknown>).id,
        result: { tools: sanitized.tools },
      };
    }

    private async handleUpstreamToolCall(
      req: Record<string, unknown>,
      sessionId: string | undefined,
      profileId: string | undefined,
      provider: UpstreamMcpServerConfig,
    ): Promise<unknown> {
      // Guard: upstream_mcp requires session context (HTTP transport only)
      if (!sessionId) {
        throw new UpstreamConnectionError(
          'upstream_mcp requires a session context (HTTP transport only)',
          { correlationId: generateCorrelationId() }
        );
      }
      const params = req.params as Record<string, unknown>;
      const toolName = params.name as string;
      const args = (params.arguments as Record<string, unknown>) || {};
      const token = this.getUpstreamToken(sessionId, profileId);

      try {
        const client = await this.getUpstreamClientFn!(sessionId, provider, token);
        const result = await client.callTool({ name: toolName, arguments: args });
        // Forward as-is including isError: true (tool-level errors are valid responses per Pitfall 4)
        return {
          jsonrpc: '2.0',
          id: (req as Record<string, unknown>).id,
          result,
        };
      } catch (error) {
        return {
          jsonrpc: '2.0',
          id: (req as Record<string, unknown>).id,
          error: mapUpstreamErrorToMcpError(error, provider.name),
        };
      }
    }

    private getUpstreamToken(sessionId: string | undefined, profileId: string | undefined): string | undefined {
      // The downstream client's auth token is used for upstream auth (pass-through model).
      // getSessionToken already exists on HttpTransport and returns session.authToken.
      if (this.httpTransport && sessionId && profileId) {
        return this.httpTransport.getSessionToken(profileId, sessionId);
      }
      return undefined;
    }
    ```

    **Change 5: Data-driven upstream error mapping (module-level function)**

    Use `instanceof` checks for typed upstream errors - NOT `constructor.name` string matching
    (which breaks under minification and when errors are re-thrown).

    IMPORTANT (addresses review concern - LOW): Define a named constant for the non-standard timeout code.
    IMPORTANT (addresses review concern - MEDIUM): Move `providerName` OUT of the error message string
    and INTO `data.providerName`. The message string is client-facing; internal infrastructure names
    (upstream provider names) should not leak to downstream clients in a security-boundary gateway.

    Add before the class definition:
    ```typescript
    /** Custom MCP timeout error code - not in MCP standard spec; used for upstream timeout distinction */
    const UPSTREAM_TIMEOUT_ERROR_CODE = -32001;

    function mapUpstreamErrorToMcpError(error: unknown, providerName: string): { code: number; message: string; data?: unknown } {
      // Use instanceof for reliable typed error dispatch
      const DATA_DRIVEN_MAPPINGS: Array<[new (...args: unknown[]) => Error, number, string]> = [
        [UpstreamConnectionError, ErrorCode.InternalError, 'Upstream connection failed'],
        [UpstreamTimeoutError, UPSTREAM_TIMEOUT_ERROR_CODE, 'Upstream request timed out'],
        [UpstreamAuthError, ErrorCode.InternalError, 'Upstream authentication failed'],
        [UpstreamMalformedResponseError, ErrorCode.InternalError, 'Upstream returned malformed response'],
      ];

      const err = error instanceof Error ? error : new Error(String(error));
      const correlationId = 'correlationId' in err ? (err as Record<string, unknown>).correlationId as string | undefined : undefined;

      for (const [ErrorClass, code, messagePrefix] of DATA_DRIVEN_MAPPINGS) {
        if (error instanceof ErrorClass) {
          return {
            code,
            message: correlationId ? `${messagePrefix} (correlation: ${correlationId})` : messagePrefix,
            data: { correlationId, providerName },
          };
        }
      }

      return {
        code: ErrorCode.InternalError,
        message: correlationId ? `Upstream error (correlation: ${correlationId})` : 'Upstream error',
        data: { correlationId, providerName },
      };
    }
    ```
