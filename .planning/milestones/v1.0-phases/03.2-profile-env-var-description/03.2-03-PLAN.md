        while (!fs.existsSync(path.join(dir, 'package.json'))) {

          dir = parent;

        return dir;

        const nextRel = html.slice(after).search(/\n\s{6}function\s/);
        if (nextRel < 0) throw new Error('end of renderDetail not found');
        return html.slice(start, after + nextRel);
