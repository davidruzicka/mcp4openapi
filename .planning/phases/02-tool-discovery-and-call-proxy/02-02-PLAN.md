    **Change 4: Add the handler methods**

    ```typescript
    private async handleUpstreamToolsList(
      req: Record<string, unknown>,
      sessionId: string | undefined,
      profileId: string | undefined,
      provider: UpstreamMcpServerConfig,
    ): Promise<unknown> {
      const token = this.getUpstreamToken(sessionId, profileId);
      const client = await this.getUpstreamClientFn!(sessionId!, provider, token);
      const result = await client.listTools();
      const sanitized = sanitizeToolList(result.tools, this.logger);
      return {
        jsonrpc: '2.0',
        id: (req as Record<string, unknown>).id,
        result: { tools: sanitized.tools },
      };
    }

    private async handleUpstreamToolCall(
      req: Record<string, unknown>,
      sessionId: string | undefined,
      profileId: string | undefined,
      provider: UpstreamMcpServerConfig,
    ): Promise<unknown> {
      const params = req.params as Record<string, unknown>;
      const toolName = params.name as string;
      const args = (params.arguments as Record<string, unknown>) || {};
      const token = this.getUpstreamToken(sessionId, profileId);

      try {
        const client = await this.getUpstreamClientFn!(sessionId!, provider, token);
        const result = await client.callTool({ name: toolName, arguments: args });
        // Forward as-is including isError: true (tool-level errors are valid responses per Pitfall 4)
        return {
          jsonrpc: '2.0',
          id: (req as Record<string, unknown>).id,
          result,
        };
      } catch (error) {
        return {
          jsonrpc: '2.0',
          id: (req as Record<string, unknown>).id,
          error: mapUpstreamErrorToMcpError(error, provider.name),
        };
      }
    }

    private getUpstreamToken(sessionId: string | undefined, profileId: string | undefined): string | undefined {
      // Get token from session data (stored at session init via X-Upstream-Authorization)
      if (this.httpTransport && sessionId && profileId) {
        const sessionData = this.httpTransport.getSessionData(profileId, sessionId);
        if (sessionData?.upstreamCredentials) {
          // Get first provider's token from the credentials map
          const firstEntry = sessionData.upstreamCredentials.entries().next().value;
          return firstEntry?.[1];
        }
      }
      return undefined;
    }
    ```

    **Change 5: Data-driven upstream error mapping (module-level function)**

    Add before the class definition or as a private static:
    ```typescript
    const UPSTREAM_ERROR_MAP: Record<string, { code: number; messagePrefix: string }> = {
      UpstreamConnectionError: { code: ErrorCode.InternalError, messagePrefix: 'Upstream connection failed' },
      UpstreamTimeoutError: { code: -32001, messagePrefix: 'Upstream request timed out' },
      UpstreamAuthError: { code: ErrorCode.InternalError, messagePrefix: 'Upstream authentication failed' },
      UpstreamMalformedResponseError: { code: ErrorCode.InternalError, messagePrefix: 'Upstream returned malformed response' },
    };

    function mapUpstreamErrorToMcpError(error: unknown, providerName: string): { code: number; message: string; data?: unknown } {
      const err = error instanceof Error ? error : new Error(String(error));
      const mapping = UPSTREAM_ERROR_MAP[err.constructor.name] ?? {
        code: ErrorCode.InternalError,
        messagePrefix: 'Upstream error',
      };
      const correlationId = 'correlationId' in err ? (err as Record<string, unknown>).correlationId : undefined;
      return {
        code: mapping.code,
        message: `${mapping.messagePrefix} (provider: ${providerName}${correlationId ? `, correlation: ${correlationId}` : ''})`,
        data: { correlationId, providerName },
      };
    }
    ```
