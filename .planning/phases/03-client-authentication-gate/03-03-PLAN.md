    ```typescript
    import type { Logger } from '../core/logger.js';
    import type { ClientAuthGateConfig } from '../types/profile.js';
    import type { AuthorizedPrincipal } from './inbound-auth-principal.js';
    import { createApiKeyStore } from './api-key-store-factory.js';
    import type { ApiKeyStore } from './api-key-store.js';
    import { ClientAuthGateError } from '../core/errors.js';

    export class ClientAuthGate {
      private readonly profileId: string;
      private readonly config: ClientAuthGateConfig;
      private readonly logger: Logger;
      private readonly apiKeyStore?: ApiKeyStore;

      constructor(profileId: string, config: ClientAuthGateConfig, logger: Logger) {
        this.profileId = profileId;
        this.config = config;
        this.logger = logger;
        if (config.api_keys) {
          this.apiKeyStore = createApiKeyStore(config.api_keys, profileId, logger);
        }
      }

      async validate(token: string | undefined): Promise<AuthorizedPrincipal | null> {
        const mode = this.config.mode ?? 'required';

        if (!token) {
          if (mode === 'optional') return null;
          this.logger.warn('Client auth gate: no token presented', { profileId: this.profileId });
          throw new ClientAuthGateError('Client authentication required: no token presented');
        }

        if (this.apiKeyStore) {
          const principal = await this.apiKeyStore.validate(token);
          if (principal) {
            // Log [API_KEY REDACTED] as explicit marker — never log the raw token value
            this.logger.info('Client auth success', {
              profileId: this.profileId,
              subject: principal.subject,
              authType: principal.authType,
              credential: '[API_KEY REDACTED]',
            });
            return principal;
          }
        }

        if (mode === 'optional') return null;

        this.logger.warn('Client auth gate: no valid identity resolved', {
          profileId: this.profileId,
          credential: '[API_KEY REDACTED]',
        });
        throw new ClientAuthGateError('Client authentication required: no valid identity resolved');
      }
    }
    ```

    - Invalid API key, mode='optional' → null returned
