    ```typescript
    /**
     * Shared OIDC discovery utility.
     * Callers convert OidcDiscoveryError to their domain error type.
     */

    // Discovery documents are small by spec (~2 KB); 64 KB is generous and prevents DoS
    // via attacker-controlled issuer URLs (consistent with JwksCache.MAX_JWKS_RESPONSE_BYTES).
    const MAX_DISCOVERY_RESPONSE_BYTES = 64 * 1024;

    export class OidcDiscoveryError extends Error {
      constructor(message: string, public readonly details?: Record<string, unknown>) {
        super(message);
        this.name = 'OidcDiscoveryError';
      }
    }

    /**
     * Resolves the JWKS URI for an OIDC provider.
     *
     * trust_mode='explicit': returns jwksUri directly (throws if absent).
     * trust_mode='discovery': fetches /.well-known/openid-configuration; validates
     *   that discovered issuer matches configured issuer to prevent JWKS hijacking.
     * Enforces a 64 KB response size limit and wraps JSON parse errors as OidcDiscoveryError
     * so callers only need to handle one error type.
     */
    export async function resolveOidcJwksUri(
      issuer: string,
      jwksUri: string | undefined,
      trustMode: 'discovery' | 'explicit',
      timeoutMs = 10_000,
    ): Promise<string> {
      if (trustMode === 'explicit') {
        if (!jwksUri) {
          throw new OidcDiscoveryError('jwks_uri is required when trust_mode is explicit', { issuer });
        }
        return jwksUri;
      }

      const discoveryUrl = new URL('/.well-known/openid-configuration', issuer).toString();
      let response: Response;
      try {
        response = await fetch(discoveryUrl, {
          headers: { accept: 'application/json' },
          signal: AbortSignal.timeout(timeoutMs),
        });
      } catch (err) {
        throw new OidcDiscoveryError('Failed to fetch OIDC discovery document', {
          issuer, discoveryUrl, cause: String(err),
        });
      }
      if (!response.ok) {
        throw new OidcDiscoveryError('OIDC discovery returned non-OK status', {
          issuer, discoveryUrl, status: response.status,
        });
      }

      // Read as text first to enforce size limit before JSON parsing
      const body = await response.text();
      if (Buffer.byteLength(body, 'utf8') > MAX_DISCOVERY_RESPONSE_BYTES) {
        throw new OidcDiscoveryError('OIDC discovery document exceeds size limit', {
          issuer, discoveryUrl, limitBytes: MAX_DISCOVERY_RESPONSE_BYTES,
        });
      }

      let metadata: { jwks_uri?: string; issuer?: string };
      try {
        metadata = JSON.parse(body) as { jwks_uri?: string; issuer?: string };
      } catch {
        throw new OidcDiscoveryError('OIDC discovery document is not valid JSON', {
          issuer, discoveryUrl,
        });
      }

      if (metadata.issuer && metadata.issuer !== issuer) {
        throw new OidcDiscoveryError('OIDC discovery issuer mismatch', {
          configured: issuer, discovered: metadata.issuer,
        });
      }
      if (typeof metadata.jwks_uri !== 'string') {
        throw new OidcDiscoveryError('OIDC discovery document missing jwks_uri', {
          issuer, discoveryUrl,
        });
      }
      return metadata.jwks_uri;
    }
    ```

    ```typescript
    import { decodeProtectedHeader, jwtVerify, errors as joseErrors } from 'jose';
    import type { Logger } from '../core/logger.js';
    import type { ClientAuthGateConfig } from '../types/profile.js';
    import type { AuthorizedPrincipal } from './inbound-auth-principal.js';
    import { JwksCache } from './jwks-cache.js';
    import { createApiKeyStore } from './api-key-store-factory.js';
    import type { ApiKeyStore } from './api-key-store.js';
    import { ClientAuthGateError } from '../core/errors.js';
    import { resolveOidcJwksUri, OidcDiscoveryError } from './oidc-discovery.js';
    const DEFAULT_ALLOWED_ALGS = ['RS256', 'RS384', 'RS512', 'ES256', 'ES384', 'ES512'] as const;

    export class ClientAuthGate {
      private readonly profileId: string;
      private readonly config: ClientAuthGateConfig;
      private readonly jwksCache: JwksCache;
      private readonly logger: Logger;
      private apiKeyStore?: ApiKeyStore;

      // jwksCache is injected from HttpTransport (shared per-transport instance, same pattern as
      // EnterpriseAuthProvider). All ClientAuthGate instances across profiles share one JWKS cache,
      // pooling JWKS keys for all configured issuers within JwksCacheOptions.maxCachedIssuers.
      constructor(profileId: string, config: ClientAuthGateConfig, jwksCache: JwksCache, logger: Logger) {
        this.profileId = profileId;
        this.config = config; // stored; not re-passed on validate() calls
        this.logger = logger;
        this.jwksCache = jwksCache; // shared instance from HttpTransport - no per-profile JWKS cache churn
        if (config.api_keys) {
          this.apiKeyStore = createApiKeyStore(config.api_keys, profileId, logger);
        }
      }

      // config is stored at construction time — not passed per-call (Issue 2 fix: DRY)
      async validate(token: string | undefined): Promise<AuthorizedPrincipal | null> {
        const mode = this.config.mode ?? 'required';

        if (!token) {
          if (mode === 'optional') return null;
          throw new ClientAuthGateError('Client authentication required: no token presented');
        }

        // D-12: JWT first (if decodable + jwt config present), then API key
        if (this.config.jwt) {
          const jwtResult = await this.tryJwt(token);
          if (jwtResult !== undefined) {
            return jwtResult; // null = JWT detection succeeded but validation failed
          }
          // jwtResult === undefined means token was not a JWT; fall through
        }

        // API key path
        if (this.apiKeyStore) {
          const keyResult = await this.apiKeyStore.validate(token);
          if (keyResult !== null) return keyResult;
        }

        // No identity resolved
        if (mode === 'required') {
          throw new ClientAuthGateError('Client authentication required: no valid identity resolved');
        }
        return null;
      }

      /**
       * Returns:
       *   AuthorizedPrincipal - JWT valid
       *   null               - JWT detected but validation failed (throws on hard errors)
       *   undefined          - token is not a JWT; caller should try API key path
       */
      private async tryJwt(token: string): Promise<AuthorizedPrincipal | null | undefined> {
        let header: { alg?: string; kid?: string };
        try {
          header = decodeProtectedHeader(token);
        } catch {
          return undefined; // not a JWT
        }
        if (!header.alg) return undefined;

        const jwtConfig = this.config.jwt!;
        const allowedAlgs = jwtConfig.allowed_algs ?? [...DEFAULT_ALLOWED_ALGS];

        if (!allowedAlgs.includes(header.alg as never)) {
          throw new ClientAuthGateError('Client JWT uses an unsupported signing algorithm', { alg: header.alg });
        }
        if (header.alg === 'none') {
          throw new ClientAuthGateError('Unsigned client JWTs are not accepted');
        }

        const jwksUri = await this.resolveJwksUri(jwtConfig);
        const resolver = await this.jwksCache.getResolver(jwtConfig.issuer, jwksUri, header.kid);

        try {
          const { payload } = await jwtVerify(token, resolver, {
            issuer: jwtConfig.issuer,
            audience: jwtConfig.audience,
            algorithms: allowedAlgs as string[],
            clockTolerance: jwtConfig.clock_skew_seconds ?? 60,
          });

          const sub = typeof payload.sub === 'string' ? payload.sub : undefined;
          if (!sub) {
            throw new ClientAuthGateError('Client JWT missing sub claim');
          }
          const expiresAt = typeof payload.exp === 'number' ? payload.exp * 1000 : undefined;
          const scopes = typeof payload.scope === 'string' ? payload.scope.split(' ') : [];

          return {
            authType: 'oauth',
            profileId: this.profileId,
            issuer: jwtConfig.issuer,
            subject: sub,
            scopes,
            expiresAt,
          };
        } catch (err) {
          if (err instanceof ClientAuthGateError) throw err;
          if (err instanceof joseErrors.JWTExpired) {
            throw new ClientAuthGateError('Client JWT has expired');
          }
          if (err instanceof joseErrors.JWTClaimValidationFailed) {
            throw new ClientAuthGateError('Client JWT claim validation failed', { reason: err.message });
          }
          if (err instanceof joseErrors.JWSSignatureVerificationFailed) {
            throw new ClientAuthGateError('Client JWT signature verification failed');
          }
          // Other jose errors: invalid token structure
          throw new ClientAuthGateError('Client JWT validation failed', { reason: String(err) });
        }
      }

      private async resolveJwksUri(jwtConfig: NonNullable<ClientAuthGateConfig['jwt']>): Promise<string> {
        // Delegates to shared oidc-discovery utility; converts OidcDiscoveryError to ClientAuthGateError
        const trustMode = jwtConfig.trust_mode ?? (jwtConfig.jwks_uri ? 'explicit' : 'discovery');
        try {
          return await resolveOidcJwksUri(jwtConfig.issuer, jwtConfig.jwks_uri, trustMode);
        } catch (err) {
          if (err instanceof OidcDiscoveryError) {
            throw new ClientAuthGateError(err.message, { ...err.details });
          }
          throw err;
        }
      }
    }
    ```
