    ```typescript
    import { decodeProtectedHeader, jwtVerify, errors as joseErrors } from 'jose';
    import type { Logger } from '../core/logger.js';
    import type { ClientAuthGateConfig } from '../types/profile.js';
    import type { AuthorizedPrincipal } from './inbound-auth-principal.js';
    import { JwksCache } from './jwks-cache.js';
    import { createApiKeyStore } from './api-key-store-factory.js';
    import type { ApiKeyStore } from './api-key-store.js';
    import { ClientAuthGateError, EnterpriseIssuerDiscoveryError } from '../core/errors.js';

    const DISCOVERY_SUFFIX = '/.well-known/openid-configuration';
    const DEFAULT_ALLOWED_ALGS = ['RS256', 'RS384', 'RS512', 'ES256', 'ES384', 'ES512'] as const;
    const JWKS_CACHE_OPTIONS = {
      maxCachedIssuers: 10,
      maxCachedKeys: 100,
      refreshTimeoutMs: 10_000,
      refreshBackoffMs: 60_000,
    };

    export class ClientAuthGate {
      private readonly profileId: string;
      private readonly jwksCache: JwksCache;
      private readonly logger: Logger;
      private apiKeyStore?: ApiKeyStore;

      constructor(profileId: string, config: ClientAuthGateConfig, logger: Logger) {
        this.profileId = profileId;
        this.logger = logger;
        this.jwksCache = new JwksCache(JWKS_CACHE_OPTIONS, logger);
        if (config.api_keys) {
          this.apiKeyStore = createApiKeyStore(config.api_keys, profileId, logger);
        }
      }

      async validate(
        token: string | undefined,
        config: ClientAuthGateConfig
      ): Promise<AuthorizedPrincipal | null> {
        const mode = config.mode ?? 'required';

        if (!token) {
          if (mode === 'optional') return null;
          throw new ClientAuthGateError('Client authentication required: no token presented');
        }

        // D-12: JWT first (if decodable + jwt config present), then API key
        if (config.jwt) {
          const jwtResult = await this.tryJwt(token, config);
          if (jwtResult !== undefined) {
            return jwtResult; // null = JWT detection succeeded but validation failed
          }
          // jwtResult === undefined means token was not a JWT; fall through
        }

        // API key path
        if (this.apiKeyStore) {
          const keyResult = await this.apiKeyStore.validate(token);
          if (keyResult !== null) return keyResult;
        }

        // No identity resolved
        if (mode === 'required') {
          throw new ClientAuthGateError('Client authentication required: no valid identity resolved');
        }
        return null;
      }

      /**
       * Returns:
       *   AuthorizedPrincipal - JWT valid
       *   null               - JWT detected but validation failed (throws on hard errors)
       *   undefined          - token is not a JWT; caller should try API key path
       */
      private async tryJwt(
        token: string,
        config: ClientAuthGateConfig
      ): Promise<AuthorizedPrincipal | null | undefined> {
        let header: { alg?: string; kid?: string };
        try {
          header = decodeProtectedHeader(token);
        } catch {
          return undefined; // not a JWT
        }
        if (!header.alg) return undefined;

        const jwtConfig = config.jwt!;
        const allowedAlgs = jwtConfig.allowed_algs ?? [...DEFAULT_ALLOWED_ALGS];

        if (!allowedAlgs.includes(header.alg as never)) {
          throw new ClientAuthGateError('Client JWT uses an unsupported signing algorithm', { alg: header.alg });
        }
        if (header.alg === 'none') {
          throw new ClientAuthGateError('Unsigned client JWTs are not accepted');
        }

        const jwksUri = await this.resolveJwksUri(jwtConfig);
        const resolver = await this.jwksCache.getResolver(jwtConfig.issuer, jwksUri, header.kid);

        try {
          const { payload } = await jwtVerify(token, resolver, {
            issuer: jwtConfig.issuer,
            audience: jwtConfig.audience,
            algorithms: allowedAlgs as string[],
            clockTolerance: jwtConfig.clock_skew_seconds ?? 60,
          });

          const sub = typeof payload.sub === 'string' ? payload.sub : undefined;
          if (!sub) {
            throw new ClientAuthGateError('Client JWT missing sub claim');
          }
          const expiresAt = typeof payload.exp === 'number' ? payload.exp * 1000 : undefined;
          const scopes = typeof payload.scope === 'string' ? payload.scope.split(' ') : [];

          return {
            authType: 'oauth',
            profileId: this.profileId,
            issuer: jwtConfig.issuer,
            subject: sub,
            scopes,
            expiresAt,
          };
        } catch (err) {
          if (err instanceof ClientAuthGateError) throw err;
          if (err instanceof joseErrors.JWTExpired) {
            throw new ClientAuthGateError('Client JWT has expired');
          }
          if (err instanceof joseErrors.JWTClaimValidationFailed) {
            throw new ClientAuthGateError('Client JWT claim validation failed', { reason: err.message });
          }
          if (err instanceof joseErrors.JWSSignatureVerificationFailed) {
            throw new ClientAuthGateError('Client JWT signature verification failed');
          }
          // Other jose errors: invalid token structure
          throw new ClientAuthGateError('Client JWT validation failed', { reason: String(err) });
        }
      }

      private async resolveJwksUri(jwtConfig: NonNullable<ClientAuthGateConfig['jwt']>): Promise<string> {
        const trustMode = jwtConfig.trust_mode ?? (jwtConfig.jwks_uri ? 'explicit' : 'discovery');
        if (trustMode === 'explicit') {
          if (!jwtConfig.jwks_uri) throw new ClientAuthGateError('jwks_uri required for trust_mode=explicit');
          return jwtConfig.jwks_uri;
        }
        // Discovery: fetch /.well-known/openid-configuration
        const discoveryUrl = `${jwtConfig.issuer.replace(/\/$/, '')}${DISCOVERY_SUFFIX}`;
        let resp: Response;
        try {
          resp = await fetch(discoveryUrl, { signal: AbortSignal.timeout(10_000) });
        } catch (err) {
          throw new EnterpriseIssuerDiscoveryError('Failed to fetch OIDC discovery document', { url: discoveryUrl, cause: String(err) });
        }
        if (!resp.ok) {
          throw new EnterpriseIssuerDiscoveryError('OIDC discovery returned non-OK status', { url: discoveryUrl, status: resp.status });
        }
        const doc = await resp.json();
        if (typeof doc.jwks_uri !== 'string') {
          throw new EnterpriseIssuerDiscoveryError('OIDC discovery document missing jwks_uri', { url: discoveryUrl });
        }
        return doc.jwks_uri;
      }
    }
    ```
