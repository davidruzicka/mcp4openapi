    ```typescript
    import { timingSafeEqual, createHmac, randomBytes } from 'node:crypto';
    import type { AuthorizedPrincipal } from './inbound-auth-principal.js';
    import type { ApiKeyStore } from './api-key-store.js';
    import type { InlineApiKeyEntry } from '../types/profile.js';

    export class InlineApiKeyStore implements ApiKeyStore {
      private readonly profileId: string;
      private readonly entries: InlineApiKeyEntry[];
      // Per-instance secret: HMAC-SHA256 of any two equal strings always yields equal
      // 32-byte digests, so timingSafeEqual compares equal-length buffers regardless
      // of key length — erasing length as a timing side-channel.
      private readonly hmacKey: Buffer;

      constructor(profileId: string, entries: InlineApiKeyEntry[]) {
        this.profileId = profileId;
        this.entries = entries;
        this.hmacKey = randomBytes(32);
      }

      private digest(value: string): Buffer {
        return createHmac('sha256', this.hmacKey).update(value, 'utf8').digest();
      }

      async validate(key: string): Promise<AuthorizedPrincipal | null> {
        const keyDigest = this.digest(key);
        for (const entry of this.entries) {
          const configured = process.env[entry.key_from_env];
          if (!configured) continue; // env var not set; skip silently
          // HMAC recomputed per entry per call - acceptable: InlineApiKeyStore targets small
          // deployments (typically 1-5 keys); pre-computed digests would add mutable state
          // and env-var invalidation complexity without meaningful latency benefit at this scale.
          // Both digests are always 32 bytes - timingSafeEqual is safe with no padding needed.
          if (timingSafeEqual(keyDigest, this.digest(configured))) {
            return {
              authType: 'token',
              profileId: this.profileId,
              subject: entry.subject,
              scopes: entry.scopes ?? [],
            };
          }
        }
        return null;
      }
    }
    ```
