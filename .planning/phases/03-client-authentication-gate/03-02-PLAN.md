    ```typescript
    import { timingSafeEqual, createHash } from 'node:crypto';
    import type { AuthorizedPrincipal } from './inbound-auth-principal.js';
    import type { ApiKeyStore } from './api-key-store.js';
    import type { InlineApiKeyEntry } from '../types/profile.js';

    export class InlineApiKeyStore implements ApiKeyStore {
      private readonly profileId: string;
      private readonly entries: InlineApiKeyEntry[];

      constructor(profileId: string, entries: InlineApiKeyEntry[]) {
        this.profileId = profileId;
        this.entries = entries;
      }

      async validate(key: string): Promise<AuthorizedPrincipal | null> {
        const keyBuf = Buffer.from(key, 'utf8');
        for (const entry of this.entries) {
          const configured = process.env[entry.key_from_env];
          if (!configured) continue; // env var not set; skip silently
          const configuredBuf = Buffer.from(configured, 'utf8');
          // Constant-time comparison. Pad shorter buffer to same length to avoid
          // length-dependent early exit. Even on length mismatch we run the full compare.
          const maxLen = Math.max(keyBuf.length, configuredBuf.length);
          const a = Buffer.alloc(maxLen);
          const b = Buffer.alloc(maxLen);
          keyBuf.copy(a);
          configuredBuf.copy(b);
          if (timingSafeEqual(a, b) && keyBuf.length === configuredBuf.length) {
            return {
              authType: 'token',
              profileId: this.profileId,
              subject: entry.subject,
              scopes: entry.scopes ?? [],
            };
          }
        }
        return null;
      }
    }
    ```

    ```typescript
    import type { Logger } from '../core/logger.js';
    import type { AuthorizedPrincipal } from './inbound-auth-principal.js';
    import type { ApiKeyStore } from './api-key-store.js';
    import { SSRFValidator } from '../security/ssrf-validator.js';
    import { ClientAuthGateError } from '../core/errors.js';

    const DEFAULT_BASE_URL = 'https://sasanka.seznam.net';
    const DEFAULT_TIMEOUT_MS = 5000;
    const USERS_ME_PATH = '/api/v1/users/me';

    interface SasankaUsersMeResponse {
      username: string;
      role: string;
      email: string;
      token: {
        token_id: string;
        read_only: boolean;
        expires: string;
        scope: string;
      };
    }

    export class SasankaApiKeyStore implements ApiKeyStore {
      private readonly profileId: string;
      private readonly baseUrl: string;
      private readonly timeoutMs: number;
      private readonly logger: Logger;

      constructor(
        profileId: string,
        baseUrl: string = DEFAULT_BASE_URL,
        timeoutMs: number = DEFAULT_TIMEOUT_MS,
        logger: Logger,
        ssrfValidator?: SSRFValidator
      ) {
        this.profileId = profileId;
        this.baseUrl = baseUrl.replace(/\/$/, ''); // strip trailing slash
        this.timeoutMs = timeoutMs;
        this.logger = logger;
        // SSRF validation in non-test environments
        if (process.env.NODE_ENV !== 'test' && ssrfValidator) {
          // Synchronous path: validate at construction; throw early
          // SSRFValidator.validateUrl is async — call synchronously by convention or wrap.
          // Pattern: validateUrl is called before first use in async validate().
        }
        this._ssrfValidator = ssrfValidator ?? new SSRFValidator(logger);
        this._ssrfValidated = process.env.NODE_ENV === 'test'; // skip in test
      }

      private readonly _ssrfValidator: SSRFValidator;
      private _ssrfValidated: boolean;

      async validate(token: string): Promise<AuthorizedPrincipal | null> {
        if (!this._ssrfValidated) {
          await this._ssrfValidator.validateUrl(this.baseUrl);
          this._ssrfValidated = true;
        }
        const url = `${this.baseUrl}${USERS_ME_PATH}`;
        let response: Response;
        try {
          response = await fetch(url, {
            method: 'GET',
            headers: { Authorization: `Bearer ${token}` },
            signal: AbortSignal.timeout(this.timeoutMs),
          });
        } catch (err) {
          this.logger.warn('Sasanka /users/me request failed', { error: String(err) });
          throw new ClientAuthGateError('Sasanka validation failed: network error', { cause: String(err) });
        }
        if (response.status === 401) {
          return null;
        }
        if (!response.ok) {
          this.logger.warn('Sasanka /users/me returned unexpected status', { status: response.status });
          throw new ClientAuthGateError('Sasanka validation failed: unexpected status', { status: response.status });
        }
        const body: SasankaUsersMeResponse = await response.json();
        const expiresAt = body.token?.expires ? Date.parse(body.token.expires) : undefined;
        return {
          authType: 'token',
          profileId: this.profileId,
          subject: body.username,
          scopes: body.token?.scope ? [body.token.scope] : [],
          expiresAt: expiresAt && !isNaN(expiresAt) ? expiresAt : undefined,
        };
      }
    }
    ```

    ```typescript
    import type { Logger } from '../core/logger.js';
    import type { ApiKeyStoreConfig } from '../types/profile.js';
    import type { ApiKeyStore } from './api-key-store.js';
    import { InlineApiKeyStore } from './inline-api-key-store.js';
    import { SasankaApiKeyStore } from './sasanka-api-key-store.js';
    import { ClientAuthGateError } from '../core/errors.js';

    type StoreCreator = (config: ApiKeyStoreConfig, profileId: string, logger: Logger) => ApiKeyStore;

    const STORE_REGISTRY: Record<string, StoreCreator> = {
      inline: (config, profileId) => {
        if (config.type !== 'inline') throw new ClientAuthGateError('Type mismatch in registry');
        return new InlineApiKeyStore(profileId, config.keys);
      },
      sasanka: (config, profileId, logger) => {
        if (config.type !== 'sasanka') throw new ClientAuthGateError('Type mismatch in registry');
        return new SasankaApiKeyStore(profileId, config.base_url, config.timeout_ms, logger);
      },
    };

    export function createApiKeyStore(config: ApiKeyStoreConfig, profileId: string, logger: Logger): ApiKeyStore {
      const creator = STORE_REGISTRY[config.type];
      if (!creator) {
        throw new ClientAuthGateError(`Unsupported api_keys.type: '${config.type}'`);
      }
      return creator(config, profileId, logger);
    }
    ```
